class TargetMotion:

    def __init__(
        self,
        motion_type="straight"
    ):

        self.motion_type = motion_type

        self.t = 0.0

        self.center_x = 1000
        self.center_y = 1000

        self.radius_x = 300
        self.radius_y = 200

        self.speed = 1.0

        self.x = 700
        self.y = 1000

        self.vx = 2
        self.vy = 1

    def update(self):

        self.t += 1 / FPS

        # ------------------------------------
        # STRAIGHT
        # ------------------------------------

        if self.motion_type == "straight":

            self.x += self.vx

            if self.x < 200 or self.x > 1800:
                self.vx *= -1

        # ------------------------------------
        # CIRCULAR
        # ------------------------------------

        elif self.motion_type == "circular":

            self.x = (
                self.center_x +
                self.radius_x *
                np.cos(self.speed * self.t)
            )

            self.y = (
                self.center_y +
                self.radius_y *
                np.sin(self.speed * self.t)
            )

        # ------------------------------------
        # FIGURE 8
        # ------------------------------------

        elif self.motion_type == "figure8":

            self.x = (
                self.center_x +
                self.radius_x *
                np.sin(self.speed * self.t)
            )

            self.y = (
                self.center_y +
                self.radius_y *
                np.sin(
                    self.speed *
                    self.t
                ) *
                np.cos(
                    self.speed *
                    self.t
                )
            )

        # ------------------------------------
        # RANDOM
        # ------------------------------------

        elif self.motion_type == "random":

            self.x += self.vx
            self.y += self.vy

            if self.x < 200 or self.x > 1800:
                self.vx *= -1

            if self.y < 200 or self.y > 1800:
                self.vy *= -1

        return self.x, self.y

def run_tracking_test(
    motion_type,
    duration=30
):

    target = TargetMotion(
        motion_type=motion_type,
        speed=120
    )

    camera = VirtualCamera()

    controller = WorldCameraController(
        kp=0.8,
        ki=0.01,
        kd=0.1,
        max_speed=5.0
    )

    errors = []

    total_frames = int(
        duration * FPS
    )

    for _ in range(total_frames):

        target_x, target_y = target.update(DT)

        px, py = get_target_pixel(
            target_x,
            target_y,
            camera
        )

        error_x = px - CAM_W / 2
        error_y = py - CAM_H / 2

        pan, tilt = controller.update(
            error_x,
            error_y,
            DT
        )

        move_camera(
            camera,
            pan,
            tilt,
            DT
        )

        error = np.sqrt(
            error_x**2 +
            error_y**2
        )

        errors.append(error)

    errors = np.array(errors)

    return {
        "Trajectory": motion_type,
        "Average Error (px)": np.mean(errors),
        "Maximum Error (px)": np.max(errors),
        "RMSE (px)": np.sqrt(
            np.mean(errors**2)
        ),
        "Lock Rate (%)":
            np.mean(errors <= 10) * 100
    }

# ------------------------------------------------------------
# YOLO single-inference function
# ------------------------------------------------------------

def detect_yolo_once_53h(frame):

    results = yolo_model(
        frame,
        verbose=False
    )

    if not results:
        return None

    result = results[0]

    if result.boxes is None:
        return None

    if len(result.boxes) == 0:
        return None

    confidences = (
        result.boxes.conf
        .cpu()
        .numpy()
    )

    best_index = int(
        np.argmax(confidences)
    )

    confidence = float(
        confidences[best_index]
    )

    if confidence < 0.25:
        return None

    box = (
        result.boxes.xyxy[best_index]
        .cpu()
        .numpy()
    )

    x1, y1, x2, y2 = box

    cx = (
        float(x1) + float(x2)
    ) / 2.0

    cy = (
        float(y1) + float(y2)
    ) / 2.0

    return (
        cx,
        cy,
        confidence
    )


# ------------------------------------------------------------
# Velocity conversion
# ------------------------------------------------------------

def angular_velocity_53h(
    target_x,
    target_y,
    vx,
    vy,
    camera
):

    dx = target_x - camera.x
    dy = target_y - camera.y

    pan_den = (
        RANGE**2
        + dx**2
    )

    tilt_den = (
        RANGE**2
        + dy**2
    )

    vx_angle = (
        RANGE * vx / pan_den
    ) * (180.0 / np.pi)

    vy_angle = (
        RANGE * vy / tilt_den
    ) * (180.0 / np.pi)

    return vx_angle, vy_angle
